# P0.4 — Remover hard limits silenciosos no Gestão e Auditor

**Prioridade:** P0
**Tipo:** fix
**Branch:** `fix/p0-4-hard-limits-slice`
**Esforço estimado:** 2-3 dias
**Depende de:** —

## Objetivo

Eliminar dois `slice()` ocultos que truncam silenciosamente o input dos agentes Gestão de Projetos e Auditor em obras médias para grandes. Hoje:
- `GestaoProjAgent.getUserPrompt()` faz `input.budgetItems.slice(0, 30)` — em obra de 250 itens, o cronograma é gerado vendo 12% do orçamento.
- `AuditorAgent.getUserPrompt()` faz `budgetItems.slice(0, 80)` — em obra de 250 itens, o "auditor de consistência" audita 32% do orçamento.

O segundo é especialmente grave porque a promessa comercial do produto inclui auditoria — vender auditoria que cobre um terço do orçamento, sem o usuário saber, é falha grave.

## Contexto

Os limites foram introduzidos para evitar estourar o context window quando os outputs do Orçamentista têm centenas de itens. A solução correta não é truncar, é resumir ou particionar — duas opções que valem a pena considerar:

**Opção A — Sumarização por categoria.** Agrupar `budgetItems` por categoria (ex.: "Estrutura", "Instalações", "Acabamento") e enviar agregados (`{ category, itemCount, totalCost, top5Items }`) ao agente. Mantém visão geral sem estourar tokens.

**Opção B — Chunking igual ao Orçamentista.** Dividir os itens em batches de 60 e rodar o agente N vezes, depois consolidar. Pré-existe `budgetChunking.ts` — pode ser reaproveitado.

**Opção recomendada para Gestão de Projetos:** A. O cronograma trabalha com sequência típica (demolição → estrutura → instalações → revestimento → acabamento), não com cada item individual. Categoria + agregados é suficiente.

**Opção recomendada para Auditor:** B. Auditoria de consistência precisa ver cada item para detectar duplicatas. Particionar em batches de 60-80 itens e fazer merge das validações no final.

  ```ts
  import type { BudgetItemDetail } from '../../shared/agents';

  export interface CategorySummary {
    category: string;
    itemCount: number;
    totalCost: number;
    totalQuantityByUnit: Record<string, number>;
    topItems: Array<{ description: string; quantity: number; unit: string; totalCost: number }>;
  }

  export function summarizeByCategory(
    budgetItems: BudgetItemDetail[],
    topN = 5,
  ): CategorySummary[] {
    const groups = new Map<string, BudgetItemDetail[]>();
    for (const item of budgetItems) {
      if ((item as any).isSummaryItem) continue;
      const cat = item.category || 'Sem categoria';
      if (!groups.has(cat)) groups.set(cat, []);
      groups.get(cat)!.push(item);
    }

    return Array.from(groups.entries()).map(([category, items]) => {
      const sorted = [...items].sort((a, b) => b.totalCost - a.totalCost);
      const top = sorted.slice(0, topN).map(i => ({
        description: i.description,
        quantity: i.quantity,
        unit: i.unit,
        totalCost: i.totalCost,
      }));
      const qtyByUnit: Record<string, number> = {};
      for (const i of items) {
        qtyByUnit[i.unit] = (qtyByUnit[i.unit] ?? 0) + i.quantity;
      }
      return {
        category,
        itemCount: items.length,
        totalCost: items.reduce((s, i) => s + i.totalCost, 0),
        totalQuantityByUnit: qtyByUnit,
        topItems: top,
      };
    });
  }
  ```

## Tarefas — Auditor (Opção B)

- [ ] Criar `server/agents/auditorChunking.ts`:
  ```ts
  import type { AuditorInput, AuditorOutput, AuditorValidation } from '../../shared/agents';

  const CHUNK_SIZE = 60;

  export function needsAuditorChunking(input: AuditorInput): boolean {
    const items = input.allAgentOutputs?.orcamentista?.budgetItems ?? [];
    return items.length > CHUNK_SIZE;
  }

  export function createAuditorChunkedInputs(input: AuditorInput): AuditorInput[] {
    const allItems = input.allAgentOutputs.orcamentista?.budgetItems ?? [];
    const chunks: AuditorInput[] = [];

    for (let i = 0; i < allItems.length; i += CHUNK_SIZE) {
      const slice = allItems.slice(i, i + CHUNK_SIZE);
      chunks.push({
        ...input,
        allAgentOutputs: {
          ...input.allAgentOutputs,
          orcamentista: {
            ...input.allAgentOutputs.orcamentista,
            budgetItems: slice,
          },
        },
        // marcar para o agente saber que é chunk
        ...(i === 0
          ? { _chunkInfo: { index: 1, total: Math.ceil(allItems.length / CHUNK_SIZE), isFirst: true } }
          : { _chunkInfo: { index: Math.floor(i / CHUNK_SIZE) + 1, total: Math.ceil(allItems.length / CHUNK_SIZE), isFirst: false } }),
      } as AuditorInput);
    }

    return chunks;
  }

  export function mergeAuditorOutputs(outputs: AuditorOutput[]): AuditorOutput {
    if (outputs.length === 0) {
      throw new Error('mergeAuditorOutputs: nenhum output para fazer merge');
    }
    if (outputs.length === 1) return outputs[0];

    // Validações: concatenar todas
    const validations: AuditorValidation[] = outputs.flatMap(o => o.validations ?? []);

    // Cross-checks: pegar do primeiro chunk (já tem o total) e adicionar específicos
    const crossAgentChecks = outputs[0].crossAgentChecks ?? [];

    // Score: média ponderada por número de itens validados em cada chunk
    const scores = outputs.map(o => o.validationScore ?? 0);
    const validationScore = Math.round(scores.reduce((s, x) => s + x, 0) / scores.length);

    // Critical errors / warnings: somar
    const criticalErrors = outputs.reduce((s, o) => s + (o.criticalErrors ?? 0), 0);
    const warnings = outputs.reduce((s, o) => s + (o.warnings ?? 0), 0);

    // Corrections: somar listas
    const budgetItemsToRemove = outputs.flatMap(o => o.corrections?.budgetItemsToRemove ?? []);
    const logisticsToRemove = outputs.flatMap(o => o.corrections?.logisticsToRemove ?? []);
    const totalImpact = budgetItemsToRemove.reduce((s, c) => s + c.estimatedImpact, 0)
                      + logisticsToRemove.reduce((s, c) => s + c.estimatedImpact, 0);

    // Auditseal: pior caso ganha
    const seal = outputs.some(o => o.auditSeal === 'rejected') ? 'rejected'
               : outputs.some(o => o.auditSeal === 'approved_with_warnings') ? 'approved_with_warnings'
               : 'approved';

    return {
      ...outputs[0],
      validations,
      crossAgentChecks,
      validationScore,
      criticalErrors,
      warnings,
      isValid: criticalErrors === 0,
      auditSeal: seal,
      corrections: {
        budgetItemsToRemove,
        logisticsToRemove,
        totalImpact,
        correctedDirectCost: (outputs[0].corrections?.correctedDirectCost ?? 0),
        correctedLogisticsCost: (outputs[0].corrections?.correctedLogisticsCost ?? 0),
      },
    };
  }
  ```

  ```ts
    const { needsAuditorChunking, createAuditorChunkedInputs, mergeAuditorOutputs } =
      await import('./auditorChunking');

    if (needsAuditorChunking(input)) {
      const chunks = createAuditorChunkedInputs(input);
  ```
